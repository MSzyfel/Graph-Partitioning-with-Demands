\section{Applications of the Generalized Conductance}
This section provides our main algorithmic result for GPD and HCD on general graphs.
% Below we give two main applications of \textsc{Generalized Conductance}. We show that given any approximation algorithm for \textsc{Generalized Conductance}, one can use it to obtain approximation algorithm for \textsc{Graph Partitioning with Demands} which further carries on to \textsc{Hierarchical Clustering with Demands} with only constant increase in the guarantee at each step.

% To do so, we show that a repeated use of the approximation algorithm for the \textsc{Generalized Conductance} allows us to give a bi-criteria approximation algorithm for \textsc{Graph Partitioning with Demands} for a certain range of input parameters. This range can be then extended by a careful recursive application of the algorithm. Furthermore, a similar approach can be applied to the \textsc{Hierarchical Clustering with Demands} problem, however, here we repeatedly apply the algorithm for the \textsc{Graph Partitioning with Demands} instead of the \textsc{Generalized Conductance}. All approximation-hardness results are collected in Section~\ref{sec:gpd-hardness}.

\subsection{Graph Partitioning with Demands}

We now apply the approximation algorithm for GC to obtain an approximation algorithm for \textsc{Graph Partitioning with Demands} (GPD). In this problem, we are given an instance $G=(V,E,D,c,w)$, together with a fixed constant $\rho\in(0,1)$. The goal is to find a subset of supply edges $C\subseteq E$ that minimizes $c\br{C}$ such that for every connected component $H$ of $G\setminus C$, we have $w\br{V_H}\leq B_\rho$.
Let $\OPTdem{G}{\rho}=c\br{C^*}$, where $C^*\subseteq E$ is an optimal solution.
Let $\rho<\eta<1$ be a constant. We say that $C\subseteq E$ is $\eta$-feasible if for every connected component $H$ of $G\setminus C$, it holds $w\br{V_H}\leq B_\eta$, instead of $w\br{V_H}\leq B_\rho$.
We will obtain a bi-criteria approximation algorithm for GPD that returns an $\eta$-feasible subset of edges $C$ such that $c\br{C}=\bigo_{\rho, \eta}\br{\gamma\br{n}}\cdot \OPTdem{G}{\rho}$, where $n=|V|$ and $\gamma\br{n}$ is an upper bound on the approximation ratio of the chosen \textsc{Generalized Conductance} algorithm on instances with at most $n$ vertices. This is done in two parts. Firstly, we show an algorithm for a limited range of $\eta$ and then extend it to the entirety of $\eta\in(\rho,1)$.

\subsubsection{Bi-criteria approximation algorithm for a limited range of $\eta$}
Let $
\delta_{\rho}=\min\brc{1/4,1-\rho}$ and $
\eta\in\br{\max\brc{1/2,1-\delta_{\rho}},1}$.
The analysis uses a generalization of the standard technique of dealing with \textsc{Balanced Cut}/\textsc{Graph Partitioning} problems using \textsc{Sparsest Cut} \cite{ImprovedApproximationAlgorithmsForMinimumWeightVertexSeparators,MulticommodityMaxFlowMinCutTheoremsAndTheirUseInDesigningApproximationAlgorithms}. However, due to the fact that we need to control both the demand in the components and the demand crossing the cut, a more careful consideration is required. We use a simple algorithm, characterized by the following claim.

\begin{theorem}\label{thm:gpd-bicriteria}
    For every fixed $\rho\in(0,1)$, let $\delta_{\rho}=\min\brc{1/4,1-\rho}$ and let $\gamma\br{n}$ be the approximation ratio of GC. For every fixed $\eta\in\br{\max\brc{1/2,1-\delta_{\rho}},1}$, there exists a polynomial-time algorithm that returns an $\eta$-feasible subset of edges $C$ such that
    \[
    c\br{C}=\frac{\gamma\br{n}}{(\eta+\delta_{\rho}-1)^2}\cdot \OPTdem{G}{\rho}.
    \]
\end{theorem}
\begin{proof}
    The algorithm is as follows:

    \input{algorithms/gpd_limited_range.tex}
    \begin{lemma}
    Every component $H\in G\setminus C$ is $\eta$-feasible.
    \end{lemma}
    \begin{proof}
    If $H$ is the same graph as $G'$ after the last iteration of the while loop, then by the halting condition of the loop, $w\br{V_H}\leq B_\eta$. Otherwise, $H$ was the smaller side of the cut in some iteration of the while loop, and thus $w\br{V_H}\leq W/2\leq B_\eta$, where we use the assumption $\eta\in\br{\max\brc{1/2,1-\delta_{\rho}},1}$ (in particular, $\eta\geq 1/2$). In either case $H$ is $\eta$-feasible.
    \end{proof}

    Consider an optimal solution $C^*\subseteq E$, $\OPTdem{G}{\rho}=c\br{C^*}$.
    We want to reason about $c\br{C}$.
    To do so, we consider the connected components of $G\setminus C^*$ and partition them into two vertex sets $A$ and $B$, each containing a significant fraction of the total demand as stated in the next lemma.
    \begin{lemma} \label{lem:ABpartition}
                For every instance $G=(V,E,D,c,w)$, there exists a partition of the connected components of $G\setminus C^*$ into two vertex sets $A$ and $B$ such that $w\br{A, V}\geq \delta_{\rho}\cdot W$ and $w\br{B, V}\geq \delta_{\rho}\cdot W$.
    \end{lemma}
    \begin{proof}
    We have the following cases:
    \begin{enumerate}
                \item There exists a connected component $H$ of $G\setminus C^*$ such that $w\br{V_H}\geq \delta_{\rho}\cdot W$.
                Set $A=V_H$ and $B=V\setminus V_H$. We have
                $w\br{A, V}\geq w\br{A} \geq \delta_{\rho}\cdot W$
                and 
                $w\br{B, V}=W-w\br{A}\geq \br{1-\rho}\cdot W\geq \delta_{\rho}\cdot W$,
                since $w\br{A}=w\br{V_H}\leq B_\rho$. The last inequality uses the definition $\delta_{\rho}=\min\brc{1/4,1-\rho}$.
        \item For every connected component $H$ of $G\setminus C^*$, we have $w\br{V_H}\leq \delta_{\rho}\cdot W$. We proceed in iterations. Start with $A=\emptyset$ and keep adding vertices of the subsequent components of $G\setminus C^*$ to $A$ until $w\br{A}\geq \delta_{\rho}\cdot W$. Let $H$ be the last component added to $A$ and let $A'$ be $A$ one iteration before that, i.e., $A'=A\setminus V_H$.
        Consider the two cases:
        \begin{itemize}
                        \item If $w\br{A', V_H}\geq \delta_{\rho}\cdot W$, then take a step back: set $A=A'$ and $B=V\setminus A'$. We have
            \[
                        w\br{A, V}\geq w\br{A, B} \geq w\br{A', V_H}\geq \delta_{\rho}\cdot W
            \]
                        and similarly, using $w\br{A,B}=w\br{B,A}$ for disjoint $A$ and $B$, $w\br{B, V}\geq \delta_{\rho}\cdot W$.
            \item Otherwise, $A$ stays unchanged and we set $B=V\setminus A$. We have $w\br{A, V}\geq \delta_{\rho}\cdot W$ by the halting condition of the above greedy rule, and
              \[
              w\br{B, V}
              = 
              W-w\br{A}= W-\br{w\br{A'}+w\br{A', V_H}+w\br{V_H}}
              \geq 
                            W-3\delta_{\rho}\cdot W,
              \]
            This gives $w\br{B, V}\geq \delta_{\rho}\cdot W$, because $\delta_{\rho}\leq 1/4$, again by $\delta_{\rho}=\min\brc{1/4,1-\rho}$.
        \end{itemize}
    \end{enumerate}
    This concludes the proof of the lemma.
  \end{proof}
    Take the partition $\br{A,B}$ from Lemma~\ref{lem:ABpartition}.
        Suppose that we are at some iteration when $w\br{V_{G'}}> B_\eta$. We have that
    \[
        w\br{A\cap V_{G'}, V_{G'}}\geq w\br{A, V}-w\br{V\setminus V_{G'}, V}> \delta_{\rho}\cdot W-\br{W-B_\eta}=\br{\eta+\delta_{\rho}-1}\cdot W.
    \]
        Here the halting condition $w\br{V_{G'}}> B_\eta$ implies $w\br{V\setminus V_{G'}, V}=W-w\br{V_{G'}}<W-B_\eta$.
        Symmetrically, for $B$ we get that, $w\br{B\cap V_{G'}, V_{G'}}\geq \br{\eta+\delta_{\rho}-1}\cdot W$. Since $\eta>1-\delta_{\rho}$ by assumption on $\eta$, we have $\eta+\delta_{\rho}-1>0$. Let $\br{S, \bar{S}}$ be the cut returned by the $\gamma\br{n}$-approximation algorithm for the \textsc{Generalized Conductance} in $G'$ in the considered iteration. Since $\br{A\cap V_{G'}, B\cap V_{G'}}$ is a cut of $G'$, we have
\begin{align*}
        \frac{c\br{S, \bar{S}}}{w\br{S, V_{G'}}\cdot w\br{\bar{S}, V_{G'}}}\leq \frac{\gamma\br{n} \cdot c\br{A, B}}{w\br{A\cap V_{G'}, V_{G'}}\cdot w\br{B\cap V_{G'}, V_{G'}}}
        \leq \frac{\gamma\br{n}\cdot c\br{A, B}}{\br{\eta+\delta_{\rho}-1}^2\cdot W^2}.
\end{align*}
Now, by rearranging the above inequality we get:
\begin{align*}
        c\br{S, \bar{S}} \leq 
     \frac{\gamma\br{n}\cdot c\br{A, B}}{\br{\eta+\delta_{\rho}-1}^2\cdot W^2}\cdot w\br{S, V_{G'}}\cdot w\br{\bar{S}, V_{G'}}
     \leq \frac{\gamma\br{n}\cdot \OPTdem{G}{\rho}\cdot w\br{S, V_{G'}}}{\br{\eta+\delta_{\rho}-1}^2\cdot W}
\end{align*}
since trivially we have $c\br{A, B}\leq \OPTdem{G}{\rho}$ and $w\br{\bar{S}, V_{G'}}\leq w\br{V_{G'}}\leq W$.

Let $\ell$ be the index of some iteration, let $G_{\ell}'$ be $G'$ at this iteration and let $\br{S_{\ell}, \bar{S_{\ell}}}$ be the cut at this iteration. By summing the above inequality over all iterations we get:
\begin{align*}
    c\br{C} &\leq \sum_{\ell} c\br{S_{\ell}, \bar{S_{\ell}}} \leq \frac{\gamma\br{n}\cdot \OPTdem{G}{\rho}}{\br{\eta+\delta_{\rho}-1}^2\cdot W}\cdot \sum_{\ell} w\br{S_{\ell}, V_{G'_{\ell}}} \leq \frac{\gamma\br{n}}{\br{\eta+\delta_{\rho}-1}^2}\cdot \OPTdem{G}{\rho}
\end{align*}
where the last inequality follows since the sets $S_{\ell}$ are disjoint, so $\sum_{\ell} w\br{S_{\ell}, V_{G'_{\ell}}}\leq W$.
\end{proof}

As an immediate corollary (by Thoerems~\ref{thm:tree-generalized-conductance} and~\ref{thm:generalized-conductance-approx}) we get:
\begin{corollary}
    For fixed $\rho\in(0,1)$ and $\eta\in\br{\max\brc{1/2,1-\delta_{\rho}},1}$ (where $\delta_{\rho}=\min\brc{1/4,1-\rho}$), there is a pseudo-approximation algorithm for \textsc{Graph Partitioning with Demands} which returns an $\eta$-feasible cut $C$ with the following approximation ratios:
    \[
    \bigo\br{\frac{\sqrt{\log n}}{\br{\eta+\delta_{\rho}-1}^2}}\cdot \OPTdem{G}{\rho}
    \quad\text{on general graphs,}\quad \frac{1}{\br{\eta+\delta_{\rho}-1}^2}\cdot \OPTdem{G}{\rho}
    \quad\text{on trees.}
    \]
\end{corollary}

\subsubsection{Extending the range of the parameters}
Next, we show how to extend the above procedure into a recursive algorithm that works for any choice of $0<\rho < \eta<1$. 
Since for $\rho\geq 3/4$ the previous procedure works for any $\eta\in(\rho,1)$, we focus on the case $\rho<3/4$. Then observe that $\delta_{\rho}=1/4$, and therefore we know, that the previous procedure works for any $\eta\in[3/4,1)$.
The next result shows an $\bigo_{\rho,\eta}\br{\gamma\br{n}}$-approximation for any choice of $0<\rho<\eta<1$. Note that the constants hidden in the $\bigo_{\rho,\eta}\br{\cdot}$ are quite large and we do not attempt to optimize them.

\begin{theorem}\label{thm:gpd-extension}
    Fix $0<\rho<3/4\leq \eta< 1$, and let $
        \kappa\br{\rho,\eta}=\max\brc{400,\ \frac{1}{\br{\br{1-\rho/\eta}\cdot 4/5}^2}}.$
    There exists a polynomial-time algorithm that outputs an $\eta$-feasible subset of edges $C_G$ such that $c\br{C_G}\leq d\cdot \kappa\br{\rho,\eta}\cdot \gamma\br{n}\cdot \OPTdem{G}{\rho},
    $
    where $d=\cl{\log_{5/4}\br{1/\eta}}$ and $\gamma\br{n}$ is the approximation factor of a polynomial-time algorithm for \textsc{Generalized Conductance}.
\end{theorem}
The proof is deferred to Appendix~\ref{app:gpd-extension}.

As an immediate corollary we get:
\begin{corollary}\label{cor:graph-partitioning-approximation}
    Fix $0<\rho<\eta<1$. There is a polynomial-time approximation algorithm for \textsc{Graph Partitioning with Demands} that returns an $\eta$-feasible subset of edges $C_G$ and has approximation guarantees:
    \[
    \bigo_{\rho,\eta}\br{\sqrt{\log n}}\cdot \OPTdem{G}{\rho}
    \quad\text{on general graphs,}\quad
    \bigo_{\rho,\eta}\br{1}\cdot \OPTdem{G}{\rho}
    \quad\text{on trees.}
    \]
    % In particular, for any fixed $0<\rho<\eta<1$, $\Lambda_{\rho,\eta}$ is a constant, and the above bounds are $\bigo\br{\sqrt{\log n}}$ and $\bigo\br{1}$, respectively.
\end{corollary}

\subsection{Hierarchical clustering with demands}
% \newcommand{\Dasgupta}{\tilde{c}}

In this problem one is given a graph $G=(V,E,D,c,w)$ with a supply-cost function $c\colon E\to\mathbb{N}$ and a demand function $w\colon D\to\mathbb{N}$. A \emph{clustering tree} $\cT$ is a rooted tree whose vertices are subgraphs of $G$ called \emph{clusters}, representing a laminar family of subgraphs of $G$, i.e., for any two clusters $H$ and $H'$, either $V_H\subseteq V_{H'}$, $V_{H'}\subseteq V_H$, or $V_H\cap V_{H'}=\emptyset$. We require the clustering tree to have the entire graph $G$ as its root cluster, \DDfix{any non-leaf cluster to have at least two child clusters whose vertex sets form a partition of the vertex set of the parent cluster,} and each leaf to be a singleton, i.e., a cluster with a single vertex of $G$. Therefore, there is a one-to-one correspondence between the leaves of $\cT$ and the vertices of $G$.
For $u,v\in V$, let $\operatorname{lca}_{\cT}(u,v)$ denote their lowest common ancestor in $\cT$, that is, the cluster that is farthest from the root and contains both $u$ and $v$. For a non-leaf node with cluster $H$, let $C_H$ be the supply edges whose endpoints lie in distinct child clusters of $H$. \DDfix{(For simplicity we equate the nodes with the clusters).} The objective is to minimize the total cost of separating the vertices according to the demand function, which is captured by the following function:
\begin{equation}\label{eq:Dasgupta}
    c\br{\cT}
    =\sum_{H\in\cT}
        w\br{V_H}\cdot c\br{C_H}
    =\sum_{uv\in E}c\br{uv}\cdot
        w\br{V_{\operatorname{lca}_{\cT}(u,v)}}.
\end{equation}
We refer to this problem as \textsc{Hierarchical Clustering with Demands}. 
% \DDfix{It generalizes the classical hierarchical clustering problem, also known as decision tree problem, that in turn is a generalization of the classical binary search.} Note, that a non-binary clustering tree can be refined into a binary one without increasing the cost~\cite{ACostFunctionForSimilarityBasedHierarchicalClustering}.

We parameterize our top-down algorithm by constants $0<\rho<\eta<1$. Assume that we have a polynomial-time $\gamma_{\rho,\eta}\br{n}$-approximate subroutine for \textsc{Graph Partitioning with Demands}, denoted by $\DPartition$, suppressing its parameters $\rho$ and $\eta$.

\begin{theorem}\label{thm:hc-gpd-transfer}
    Fix constants $0<\rho<\eta<1$ and suppose that a subroutine $\DPartition$ is given, where $\gamma_{\rho,\eta}(n)\geq1$. Then there exists a $\frac{\lceil1/\rho\rceil\cdot\gamma_{\rho,\eta}(n)}{1-\eta}$-approximation polynomial-time algorithm for \textsc{Hierarchical Clustering with Demands}.
\end{theorem}
\begin{proof}
    The following algorithm recursively partitions each non-singleton cluster using the subroutine for GPD and constructs the clustering tree $\cT$ in a top-down manner.
    
\medskip
\input{algorithms/hierarchical_clustering.tex}
\medskip

    Let $\cT^*$ be an optimal clustering tree, i.e., one that minimizes~\eqref{eq:Dasgupta}. For $q\geq0$, define
    \[
        E_q^*
        =\left\{
            uv\in E\colon
            w\br{V_{\operatorname{lca}_{\cT^*}(u,v)}}>q
        \right\}.
    \]
    By summing up the contribution of every supply-edge over the levels below the demand of its least common ancestor, we get that $
        c\br{\cT^*}
        =\sum_{q=0}^{W}c\br{E_q^*}.
    $

    Fix a non-singleton cluster $H$ produced by the algorithm and 
    write $r_H=w\br{V_H}$. The maximal clusters of $\cT^*$ having demand at most $\fl{\rho r_H}$ partition $V$, and by extension $V_H$, into clusters of demand at most $\fl{\rho r_H}$.
    Every supply edge of $E_H$ joining two distinct parts of this partition has its endpoints in distinct maximal clusters of $\cT^*$, so their least common ancestor in $\cT^*$ has demand greater than $\fl{\rho r_H}$. Consequently, the set $E_{\fl{\rho r_H}}^*\cap E_H$ is a feasible solution to \textsc{Graph Partitioning with Demands} on the graph $H$ with target $\rho$, so $
        c\br{C_H}\leq \gamma_{\rho,\eta}\br{\spr{V_H}} \cdot c\br{E_{\fl{\rho r_H}}^*\cap E_H}.$
    Let $s_H$ be the largest demand among the children of $H$. Since the subroutine returns an $\eta$-feasible solution, $s_H\leq\eta \cdot r_H$. Hence
    \[
        r_H\cdot c\br{C_H}
        \leq\frac{\gamma_{\rho,\eta}\br{\spr{V_H}}}{1-\eta}
            \cdot\br{r_H-s_H}\cdot c\br{E_{\fl{\rho r_H}}^*\cap E_H} \leq\frac{\gamma_{\rho,\eta}\br{\spr{V_H}}}{1-\eta}\cdot
            \sum_{t=s_H+1}^{r_H}c\br{E_{\fl{\rho t}}^*\cap E_H}
    \]
    where the first inequality is by $r_H-s_H\geq(1-\eta)r_H$. For every integer $t\leq r_H$, $E_{\fl{\rho r_H}}^*\subseteq E_{\fl{\rho t}}^*$, summing this inclusion over $t=s_H+1,\ldots,r_H$ gives the second inequality.

    For a fixed $t$, the clusters satisfying $s_H<t\leq r_H$ have pairwise disjoint induced edge sets. Indeed, if one such cluster were a descendant of another, it would lie in a non-singleton child of the ancestor and would have demand at most the ancestor's value $s_H<t$, contradicting $t\leq r_H$ for the descendant. Summing over all clusters, we conclude that
    \begin{align*}
        c\br{\cT}
        \leq\frac{\gamma_{\rho,\eta}(n)}{1-\eta}\cdot
            \sum_{t=0}^{W}c\br{E_{\fl{\rho t}}^*}
            \leq\frac{\lceil1/\rho\rceil\cdot \gamma_{\rho,\eta}(n)}{1-\eta}\cdot
            \sum_{t=0}^{W}c\br{E_{t}^*}
        =\frac{\lceil1/\rho\rceil\cdot\gamma_{\rho,\eta}(n)}{1-\eta}\cdot
            c\br{\cT^*},
    \end{align*}
    where the first inequality uses the disjointness of clusters on each level. For each $q$, at most $\lceil1/\rho\rceil$ integers $t$ satisfy $\fl{\rho t}=q$, so reindexing gives the second inequality.
\end{proof}

For the approximation bounds below, we apply Theorem~\ref{thm:hc-gpd-transfer} with $\rho=1/2$.

\begin{corollary}\label{cor:hc-general-bounds}
    There is an $\bigo\br{\sqrt{\log n}}$-approximation algorithm for \textsc{Hierarchical Clustering with Demands} on general graphs and an $\bigo\br{1}$-approximation algorithm on trees.
\end{corollary}
