\begin{algorithm}[H]
    \caption{Recursive extension to the full parameter range}
    \label{alg:recursive-partitioning}
    \textbf{Input} A component $H$, a constant $\eta$, and the algorithm from Theorem~\ref{thm:gpd-bicriteria}\;
    \textbf{Output} A $C_H\subseteq E_H$ such that each component of $H\setminus C_H$ has demand at most $\eta\cdot W$\;
    \If{$w\br{V_H}\leq \eta\cdot W$}{
        \textbf{return} $\emptyset$\;
    }
    $\eta_0\gets\max\brc{4/5,\,\eta\cdot W/w\br{V_H}}$\;
    Run Algorithm~\ref{algo:partitioning-limited-range} on $H$ with parameter $\eta_0$. Let $C$ be the returned supply-edge set\;
    $C_H\gets C$\;
    \ForEach{connected component $H'$ of $H\setminus C$}{
        Recursively partition $H'$ and add the returned supply-edge set $C_{H'}$ to $C_H$\;
    }
    \textbf{return} $C_H$\;
\end{algorithm}
