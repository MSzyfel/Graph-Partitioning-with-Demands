\begin{algorithm}[H]
    \caption{Recursive extension to the full parameter range}
    \textbf{Input} A component $H$, global budgets $B_\rho=\rho\cdot W$ and $B_\eta=\eta\cdot W$, and the algorithm from Theorem~\ref{thm:gpd-bicriteria} (Algorithm~\ref{algo:partitioning-limited-range})\;
    \textbf{Output} An $\eta$-feasible supply-edge set $C_H$\;
    \If{$w\br{V_H}\leq B_\eta$}{
        \textbf{return} $\emptyset$\;
    }
    $\eta_0\gets\max\brc{4/5,\,B_\eta/w\br{V_H}}$\;
    Run Algorithm~\ref{algo:partitioning-limited-range} on $H$ with parameter $\eta_0$. Let $C$ be the returned supply-edge set\;
    $C_H\gets C$\;
    \ForEach{connected component $H'$ of $H\setminus C$}{
        Recursively partition $H'$ and add the returned supply-edge set $C_{H'}$ to $C_H$\;
    }
    \textbf{return} $C_H$\;
\end{algorithm}
