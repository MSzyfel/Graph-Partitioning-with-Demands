\begin{algorithm}[H]
    \caption{Top-down hierarchical clustering with demands}
    \label{algo:hclustering}
    \textbf{Input} Graph $G$, parameter $\eta\in(1/2,1)$, and the subroutine $\DPartition$ satisfying~\eqref{eq:hc-gpd-subroutine}\;
    \textbf{Output} A clustering tree $\cT$\;
    Initialize $\cT$ with root cluster $G$\;
    \textbf{Procedure} $\textsc{BuildHierarchy}(H)$\;
    \Indp
            \If{$w\br{V_H}=0$}{
                Make the vertices of $H$ singleton children\;
            }
            \Else{
                Apply a $\gamma(\eta,n)$-approximation algorithm $\DPartition$  on $H$\;

                Make the components of $H\setminus C_H$ the children of $H$\;
            }
            \ForEach{child $K$ of $H$}{
                $\textsc{BuildHierarchy}(K)$\;
            }
    \Indm
    $\textsc{BuildHierarchy}(G)$\;
    \textbf{return} $\cT$\;
\end{algorithm}
