      opacity=0.35
    ] table[x=rho,y=eta,z=ratio] {generated/tree_approximation_ratio_3d_cap.dat};
    % Above eta=(1+rho)/2, centroid coverage guarantees ratio 1.
    \addplot3[
      surf,
      mesh/ordering=x varies,
      mesh/rows=51,
      mesh/cols=256,
      point meta=z,
      opacity=0.35
    ] table[x=rho,y=eta,z=ratio] {generated/tree_approximation_ratio_3d_direct.dat};
